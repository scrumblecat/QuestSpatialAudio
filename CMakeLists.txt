cmake_minimum_required(VERSION 3.22)
option(QSA_TEST_ONLY "Build standalone Android DSP tests without mod dependencies" OFF)

if(NOT QSA_TEST_ONLY)
    include(${CMAKE_CURRENT_LIST_DIR}/qpm_defines.cmake)
    set(ANDROID_PLATFORM 29)
endif()

project(quest-spatial-audio VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(QSA_TEST_ONLY)
    enable_testing()
    add_executable(qsa-dsp-tests tests/renderer_test.cpp src/renderer.cpp src/upmixer.cpp src/hrtf_data.cpp)
    target_include_directories(qsa-dsp-tests PRIVATE include)
    target_compile_options(qsa-dsp-tests PRIVATE -O3 -Wall -Wextra -Werror)
    add_test(NAME renderer COMMAND qsa-dsp-tests)
    return()
endif()

set(COMPILE_ID quest-spatial-audio)
add_library(${COMPILE_ID} SHARED src/main.cpp src/music_filter.cpp src/renderer.cpp src/upmixer.cpp src/hrtf_data.cpp)
target_include_directories(${COMPILE_ID} PRIVATE include)
target_compile_options(${COMPILE_ID} PRIVATE -O3 -frtti -fexceptions -fvisibility=hidden -Wall -Wextra)
target_compile_definitions(${COMPILE_ID} PRIVATE MOD_ID="quest-spatial-audio" VERSION="${PROJECT_VERSION}")
target_link_libraries(${COMPILE_ID} PRIVATE log)
include(${CMAKE_CURRENT_LIST_DIR}/extern.cmake)
file(GLOB inline_hook_c CONFIGURE_DEPENDS "${EXTERN_DIR}/includes/beatsaber-hook/shared/inline-hook/*.c")
file(GLOB inline_hook_cpp CONFIGURE_DEPENDS "${EXTERN_DIR}/includes/beatsaber-hook/shared/inline-hook/*.cpp")
target_sources(${COMPILE_ID} PRIVATE ${inline_hook_c} ${inline_hook_cpp})
target_link_options(${COMPILE_ID} PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
